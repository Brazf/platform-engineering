# JWT Authentication System Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implementar autenticação JWT completa — registro com bcrypt, login com geração de token, middleware de validação, erros padronizados.

**Architecture:** UseCase contém toda a lógica de negócio (hash, JWT). Handler apenas valida payload e mapeia erros para HTTP status codes. Middleware extrai e valida o token, injetando `userID` e `role` no contexto Gin. Nenhuma lógica de auth toca o Handler ou o Middleware além dessas responsabilidades.

**Tech Stack:** golang-jwt/jwt/v5, golang.org/x/crypto/bcrypt, Gin `ShouldBindJSON` com tags `binding`.

---

## Decisões de design

- **`Login` retorna `*LoginOutput`** (token + user) em vez de só `string`, porque o handler precisa de ambos para montar `{ token, user }` sem violar a separação de camadas (sem o handler importar JWT ou acessar repositório).
- **Sentinel errors exportados** (`ErrEmailAlreadyInUse`, `ErrInvalidCredentials`, `ErrInvalidRole`) para que o handler possa fazer `errors.Is` sem strings mágicas.
- **Middleware em `middleware/auth.go`** (arquivo existente) — não faz sentido criar `auth_middleware.go` separado para o mesmo package; o spec nomeia o arquivo como referência, não como regra de nomenclatura Go.
- **`RequireRole`** implementado de verdade: lê `role` do contexto, aborta com 403 se não autorizado.
- **bcrypt.DefaultCost** — custo 10, suficiente para ambiente acadêmico; em produção usar 12.

```go
package usecase

import (
	"context"
	"errors"
	"fmt"
	"time"

	"github.com/golang-jwt/jwt/v5"
	"github.com/google/uuid"
	"github.com/proxlab/backend/internal/domain/entity"
	"github.com/proxlab/backend/internal/domain/repository"
	"golang.org/x/crypto/bcrypt"
)

var (
	ErrEmailAlreadyInUse  = errors.New("email já está em uso")
	ErrInvalidCredentials = errors.New("email ou senha inválidos")
	ErrInvalidRole        = errors.New("perfil inválido: use 'aluno' ou 'pesquisador'")
)

type LoginOutput struct {
	Token string
	User  *entity.User
}

type jwtClaims struct {
	UserID string `json:"user_id"`
	Role   string `json:"role"`
	jwt.RegisteredClaims
}

type AuthUseCase struct {
	userRepo  repository.UserRepository
	jwtSecret string
	expiryH   int
}

func NewAuthUseCase(repo repository.UserRepository, jwtSecret string, expiryH int) *AuthUseCase {
	return &AuthUseCase{userRepo: repo, jwtSecret: jwtSecret, expiryH: expiryH}
}

// Register cria um novo usuário com senha hasheada.
func (uc *AuthUseCase) Register(email, password, role string) (*entity.User, error) {
	user := &entity.User{Role: entity.UserRole(role)}
	if !user.ValidateRole() {
		return nil, ErrInvalidRole
	}

	existing, err := uc.userRepo.FindByEmail(context.Background(), email)
	if err != nil {
		return nil, fmt.Errorf("erro ao verificar email: %w", err)
	}
	if existing != nil {
		return nil, ErrEmailAlreadyInUse
	}

	hash, err := bcrypt.GenerateFromPassword([]byte(password), bcrypt.DefaultCost)
	if err != nil {
		return nil, fmt.Errorf("erro ao processar senha: %w", err)
	}

	user.ID = uuid.New()
	user.Email = email
	user.PasswordHash = string(hash)

	if err := uc.userRepo.Create(context.Background(), user); err != nil {
		return nil, fmt.Errorf("erro ao criar usuário: %w", err)
	}

	return user, nil
}

// Login verifica credenciais e retorna token JWT e dados do usuário.
func (uc *AuthUseCase) Login(email, password string) (*LoginOutput, error) {
	user, err := uc.userRepo.FindByEmail(context.Background(), email)
	if err != nil {
		return nil, fmt.Errorf("erro ao buscar usuário: %w", err)
	}
	if user == nil {
		return nil, ErrInvalidCredentials
	}

	if err := bcrypt.CompareHashAndPassword([]byte(user.PasswordHash), []byte(password)); err != nil {
		return nil, ErrInvalidCredentials
	}

	expiry := time.Now().Add(time.Duration(uc.expiryH) * time.Hour)
	token := jwt.NewWithClaims(jwt.SigningMethodHS256, jwtClaims{
		UserID: user.ID.String(),
		Role:   string(user.Role),
		RegisteredClaims: jwt.RegisteredClaims{
			Subject:   user.ID.String(),
			IssuedAt:  jwt.NewNumericDate(time.Now()),
			ExpiresAt: jwt.NewNumericDate(expiry),
		},
	})

	signed, err := token.SignedString([]byte(uc.jwtSecret))
	if err != nil {
		return nil, fmt.Errorf("erro ao assinar token: %w", err)
	}

	return &LoginOutput{Token: signed, User: user}, nil
}
```

package handler

import (
	"errors"
	"net/http"

	"github.com/gin-gonic/gin"
	"github.com/proxlab/backend/internal/usecase"
)

type AuthHandler struct {
	authUC *usecase.AuthUseCase
}

func NewAuthHandler(authUC *usecase.AuthUseCase) *AuthHandler {
	return &AuthHandler{authUC: authUC}
}

type registerRequest struct {
	Email    string `json:"email"    binding:"required,email"`
	Password string `json:"password" binding:"required,min=8"`
	Role     string `json:"role"     binding:"required,oneof=aluno pesquisador"`
}

type loginRequest struct {
	Email    string `json:"email"    binding:"required,email"`
	Password string `json:"password" binding:"required"`
}

// Register godoc
//
// POST /api/auth/register
// Body:   { "email": "joao@ifpe.edu.br", "password": "senha1234", "role": "aluno" }
// 201:    { "id": "uuid", "email": "joao@ifpe.edu.br", "role": "aluno" }
// 400:    { "error": "Key: 'registerRequest.Email' Error:Field validation for 'Email' failed on the 'email' tag" }
// 409:    { "error": "email já está em uso" }
// 422:    { "error": "perfil inválido: use 'aluno' ou 'pesquisador'" }
func (h *AuthHandler) Register(c *gin.Context) {
	var req registerRequest
	if err := c.ShouldBindJSON(&req); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
		return
	}

	user, err := h.authUC.Register(req.Email, req.Password, req.Role)
	if err != nil {
		switch {
		case errors.Is(err, usecase.ErrEmailAlreadyInUse):
			c.JSON(http.StatusConflict, gin.H{"error": err.Error()})
		case errors.Is(err, usecase.ErrInvalidRole):
			c.JSON(http.StatusUnprocessableEntity, gin.H{"error": err.Error()})
		default:
			c.JSON(http.StatusInternalServerError, gin.H{"error": "erro interno do servidor"})
		}
		return
	}

	c.JSON(http.StatusCreated, gin.H{
		"id":    user.ID,
		"email": user.Email,
		"role":  user.Role,
	})
}

// Login godoc
//
// POST /api/auth/login
// Body:   { "email": "joao@ifpe.edu.br", "password": "senha1234" }
// 200:    { "token": "eyJ...", "user": { "id": "uuid", "email": "joao@ifpe.edu.br", "role": "aluno" } }
// 400:    { "error": "Key: 'loginRequest.Email' Error:..." }
// 401:    { "error": "email ou senha inválidos" }
func (h *AuthHandler) Login(c *gin.Context) {
	var req loginRequest
	if err := c.ShouldBindJSON(&req); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
		return
	}

	out, err := h.authUC.Login(req.Email, req.Password)
	if err != nil {
		if errors.Is(err, usecase.ErrInvalidCredentials) {
			c.JSON(http.StatusUnauthorized, gin.H{"error": err.Error()})
			return
		}
		c.JSON(http.StatusInternalServerError, gin.H{"error": "erro interno do servidor"})
		return
	}

	c.JSON(http.StatusOK, gin.H{
		"token": out.Token,
		"user": gin.H{
			"id":    out.User.ID,
			"email": out.User.Email,
			"role":  out.User.Role,
		},
	})
}

package middleware

import (
	"net/http"
	"strings"

	"github.com/gin-gonic/gin"
	"github.com/golang-jwt/jwt/v5"
)

const (
	ContextKeyUserID = "userID"
	ContextKeyRole   = "role"
)

type jwtClaims struct {
	UserID string `json:"user_id"`
	Role   string `json:"role"`
	jwt.RegisteredClaims
}

// Auth valida o JWT do header Authorization: Bearer <token>.
// Injeta userID (string) e role (string) no contexto Gin.
// Retorna 401 se token ausente, inválido ou expirado.
func Auth(jwtSecret string) gin.HandlerFunc {
	return func(c *gin.Context) {
		authHeader := c.GetHeader("Authorization")
		if !strings.HasPrefix(authHeader, "Bearer ") {
			c.AbortWithStatusJSON(http.StatusUnauthorized, gin.H{"error": "token de autenticação ausente"})
			return
		}

		rawToken := strings.TrimPrefix(authHeader, "Bearer ")

		token, err := jwt.ParseWithClaims(rawToken, &jwtClaims{}, func(t *jwt.Token) (any, error) {
			if _, ok := t.Method.(*jwt.SigningMethodHMAC); !ok {
				return nil, jwt.ErrSignatureInvalid
			}
			return []byte(jwtSecret), nil
		})
		if err != nil || !token.Valid {
			c.AbortWithStatusJSON(http.StatusUnauthorized, gin.H{"error": "token inválido ou expirado"})
			return
		}

		claims, ok := token.Claims.(*jwtClaims)
		if !ok {
			c.AbortWithStatusJSON(http.StatusUnauthorized, gin.H{"error": "token com formato inválido"})
			return
		}

		c.Set(ContextKeyUserID, claims.UserID)
		c.Set(ContextKeyRole, claims.Role)
		c.Next()
	}
}

// RequireRole aborta com 403 se o role do usuário autenticado não estiver na lista permitida.
// Deve ser usado após o middleware Auth.
func RequireRole(roles ...string) gin.HandlerFunc {
	allowed := make(map[string]struct{}, len(roles))
	for _, r := range roles {
		allowed[r] = struct{}{}
	}

	return func(c *gin.Context) {
		role, _ := c.Get(ContextKeyRole)
		if _, ok := allowed[role.(string)]; !ok {
			c.AbortWithStatusJSON(http.StatusForbidden, gin.H{"error": "acesso não autorizado para este perfil"})
			return
		}
		c.Next()
	}
}
